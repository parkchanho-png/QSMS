// Analyze Event (실제 AI 백엔드 연동) analyzeBtn.addEventListener('click', async () => { if (!selectedFile) return; emptyState.classList.add('hidden'); resultDashboard.classList.add('hidden'); loadingState.classList.remove('hidden'); analyzeBtn.disabled = true; try { const formData = new FormData(); formData.append('image', selectedFile); // ⚠️ 아래 URL을 생성하신 Cloudflare Worker URL로 변경해 주세요. const response = await fetch('https://label-compliance-checker-backend.본인아이디.workers.dev/api/analyze', { method: 'POST', body: formData }); const data = await response.json(); if (data.success && data.result) { // AI 분석 결과(JSON) 파싱 및 화면 표시 let parsedResult; try { parsedResult = typeof data.result === 'string' ? JSON.parse(data.result) : data.result; } catch (e) { parsedResult = data.result; } renderRealResults(parsedResult); } else { alert('분석에 실패했습니다. 다시 시도해 주세요.'); } } catch (err) { console.error(err); alert('서버 통신 중 오류가 발생했습니다.'); } finally { loadingState.classList.add('hidden'); resultDashboard.classList.remove('hidden'); analyzeBtn.disabled = false; } }); // 동적 결과 출력 함수 function renderRealResults(res) { const banner = document.getElementById('statusBanner'); const statusIcon = document.getElementById('statusIcon'); const statusTitle = document.getElementById('statusTitle'); const statusDesc = document.getElementById('statusDesc'); const isCompliant = res.is_compliant; if (isCompliant) { banner.className = 'p-4 rounded-xl border border-emerald-200 bg-emerald-50 text-emerald-900 flex items-center justify-between'; statusIcon.className = 'p-2 rounded-lg bg-emerald-600 text-white'; statusIcon.innerHTML = ``; statusTitle.textContent = '법령 위반 사항 없음'; statusDesc.textContent = '필수 한글표시사항 및 표기 기준을 준수하고 있습니다.'; } else { banner.className = 'p-4 rounded-xl border border-red-200 bg-red-50 text-red-900 flex items-center justify-between'; statusIcon.className = 'p-2 rounded-lg bg-red-600 text-white'; statusIcon.innerHTML = ``; statusTitle.textContent = '법령 위반/검토 필요 항목 발견'; statusDesc.textContent = '필수 항목 누락 또는 표현 검토가 필요한 항목이 있습니다.'; } // 필수 항목 점검 const fieldsContainer = document.getElementById('fieldsContainer'); const fields = res.required_fields || []; fieldsContainer.innerHTML = fields.map(f => ` ${f.name} `).join(''); // 위반/주의 항목 목록 const violationsList = document.getElementById('violationsList'); const violations = res.violations || []; if (violations.length === 0) { violationsList.innerHTML = `

특이 위반 사항이 감지되지 않았습니다.

`; } else { violationsList.innerHTML = violations.map(v => `
"${v.word || '검출 항목'}" 점검 필요

사유: ${v.issue || '내용 확인 필요'}

관련 법령: ${v.law || '관계법령'}

${v.guide || ''}

`).join(''); } lucide.createIcons(); }
